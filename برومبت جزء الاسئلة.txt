## 2️⃣ مكون واجهة المستخدم والتفاعل (`src/app/tools/prerequisites/quiz/page.tsx`)

إليك المكون التفاعلي كاملاً المبني بـ **Next.js 16 Client Component** وتنسيقات HSL وCairo Font السلسة:

```tsx
// src/app/tools/prerequisites/quiz/page.tsx
"use client";

import { useState } from "react";
import Link from "next/link";
import { AppShell } from "@/components/layout/app-shell";
import { diagnosticData, QuizQuestion } from "@/data/prerequisites-quiz-data";

export default function PrerequisitesQuizPage() {
  const [selectedSubject, setSelectedSubject] = useState<"math" | "physics" | "science">("math");
  const [mode, setMode] = useState<"quiz" | "exercise">("quiz");
  
  // حالة الاختبار التفاعلي
  const [currentIndex, setCurrentIndex] = useState(0);
  const [selectedAnswer, setSelectedAnswer] = useState<number | boolean | null>(null);
  const [showExplanation, setShowExplanation] = useState(false);
  const [score, setScore] = useState(0);
  const [isCompleted, setIsCompleted] = useState(false);
  const [showSolution, setShowSolution] = useState(false);

  const activeDiagnostic = diagnosticData[selectedSubject];
  const currentQuestion: QuizQuestion = activeDiagnostic.questions[currentIndex];

  const handleSubjectChange = (subject: "math" | "physics" | "science") => {
    setSelectedSubject(subject);
    resetQuiz();
  };

  const resetQuiz = () => {
    setCurrentIndex(0);
    setSelectedAnswer(null);
    setShowExplanation(false);
    setScore(0);
    setIsCompleted(false);
    setShowSolution(false);
  };

  const handleAnswerSelect = (answer: number | boolean) => {
    if (selectedAnswer !== null) return; // منع التغيير بعد الاختيار
    setSelectedAnswer(answer);
    setShowExplanation(true);

    if (answer === currentQuestion.correctAnswer) {
      setScore((prev) => prev + 1);
    }
  };

  const handleNextQuestion = () => {
    if (currentIndex + 1 < activeDiagnostic.questions.length) {
      setCurrentIndex((prev) => prev + 1);
      setSelectedAnswer(null);
      setShowExplanation(false);
    } else {
      setIsCompleted(true);
    }
  };

  return (
    <AppShell>
      <div className="back-link-wrapper" style={{ margin: "1rem 0" }}>
        <Link className="back-link" href="/tools/prerequisites">
          ← العودة للمكتسبات القبلية
        </Link>
      </div>

      {/* Heading Section */}
      <section className="subject-page-heading" style={{ marginBottom: "1.5rem" }}>
        <div>
          <p className="eyebrow">تشخيص المكتسبات القبلية 🎯</p>
          <h1>اختبار المكتسبات - 3 ثانوي</h1>
          <p style={{ marginTop: "0.5rem", color: "var(--ink-700)" }}>
            اختبر معارفك السابقة من 1 و 2 ثانوي في المواد العلمية الثلاث لتحديد نقاط قوتك وتدارك ثغراتك قبل بداية العام الدراسي[cite: 4].
          </p>
        </div>
        <span className={`subject-hero-icon subject-icon-${activeDiagnostic.color}`} aria-hidden="true">
          {activeDiagnostic.icon}
        </span>
      </section>

      {/* Navigation Controls: Subject Selector */}
      <div style={{ display: "flex", gap: "0.5rem", flexWrap: "wrap", marginBottom: "1rem" }}>
        {(["math", "physics", "science"] as const).map((key) => {
          const item = diagnosticData[key];
          const isActive = selectedSubject === key;
          return (
            <button
              key={key}
              onClick={() => handleSubjectChange(key)}
              style={{
                minHeight: "48px",
                padding: "0.6rem 1.2rem",
                borderRadius: "0.65rem",
                border: "1px solid var(--line)",
                backgroundColor: isActive ? "var(--blue-700)" : "var(--surface)",
                color: isActive ? "#ffffff" : "var(--ink-900)",
                fontWeight: "800",
                cursor: "pointer",
                display: "flex",
                alignItems: "center",
                gap: "0.5rem",
                transition: "all 0.2s ease"
              }}
            >
              <span>{item.icon}</span>
              <span>{item.subjectName}</span>
            </button>
          );
        })}
      </div>

      {/* Mode Switcher Tabs */}
      <div className="subject-tabs" style={{ marginBottom: "1.5rem" }}>
        <button
          className={`subject-tab ${mode === "quiz" ? "is-active" : ""}`}
          onClick={() => setMode("quiz")}
          type="button"
          style={{ minHeight: "48px" }}
        >
          ⏱ الاختبار التفاعلي (10 أسئلة)[cite: 4]
        </button>
        <button
          className={`subject-tab ${mode === "exercise" ? "is-active" : ""}`}
          onClick={() => setMode("exercise")}
          type="button"
          style={{ minHeight: "48px" }}
        >
          📝 التمرين الشامل المحلول[cite: 4]
        </button>
      </div>

      {/* MODE 1: INTERACTIVE QUIZ */}
      {mode === "quiz" && (
        <section className="quiz-container" style={{ maxWidth: "62rem" }}>
          {!isCompleted ? (
            <div className="calculator-card" style={{ padding: "1.5rem" }}>
              {/* Progress Bar */}
              <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: "0.75rem", fontSize: "0.85rem", fontWeight: "bold" }}>
                <span>السؤال {currentIndex + 1} من {activeDiagnostic.questions.length}</span>
                <span className="badge">النتيجة الحالية: {score}</span>
              </div>
              <div style={{ width: "100%", height: "8px", backgroundColor: "var(--line)", borderRadius: "999px", overflow: "hidden", marginBottom: "1.5rem" }}>
                <div style={{ width: `${((currentIndex + 1) / activeDiagnostic.questions.length) * 100}%`, height: "100%", backgroundColor: "var(--green-600)", transition: "width 0.3s ease" }} />
              </div>

              {/* Question Text */}
              <h2 style={{ fontSize: "1.15rem", color: "var(--blue-900)", marginBottom: "1.25rem", lineHeight: "1.6" }}>
                {currentQuestion.questionText}
              </h2>

              {/* MCQ Options */}
              {currentQuestion.type === "mcq" && currentQuestion.options && (
                <div style={{ display: "grid", gap: "0.75rem", marginBottom: "1.5rem" }}>
                  {currentQuestion.options.map((opt, idx) => {
                    let optionStyle = {
                      backgroundColor: "var(--canvas)",
                      borderColor: "var(--line)",
                      color: "var(--ink-900)"
                    };

                    if (selectedAnswer !== null) {
                      if (idx === currentQuestion.correctAnswer) {
                        optionStyle = { backgroundColor: "var(--green-100)", borderColor: "var(--green-600)", color: "var(--green-800)" };
                      } else if (selectedAnswer === idx) {
                        optionStyle = { backgroundColor: "#fff0ef", borderColor: "#c24747", color: "#c24747" };
                      }
                    }

                    return (
                      <button
                        key={idx}
                        onClick={() => handleAnswerSelect(idx)}
                        disabled={selectedAnswer !== null}
                        style={{
                          minHeight: "52px",
                          padding: "0.85rem 1.25rem",
                          borderRadius: "0.75rem",
                          border: "1px solid",
                          textAlign: "right",
                          fontSize: "0.95rem",
                          fontWeight: "700",
                          cursor: selectedAnswer === null ? "pointer" : "default",
                          ...optionStyle,
                          transition: "all 0.15s ease"
                        }}
                      >
                        {opt}
                      </button>
                    );
                  })}
                </div>
              )}

              {/* Boolean Options */}
              {currentQuestion.type === "boolean" && (
                <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: "1rem", marginBottom: "1.5rem" }}>
                  {[true, false].map((boolVal) => {
                    let boolStyle = {
                      backgroundColor: "var(--canvas)",
                      borderColor: "var(--line)",
                      color: "var(--ink-900)"
                    };

                    if (selectedAnswer !== null) {
                      if (boolVal === currentQuestion.correctAnswer) {
                        boolStyle = { backgroundColor: "var(--green-100)", borderColor: "var(--green-600)", color: "var(--green-800)" };
                      } else if (selectedAnswer === boolVal) {
                        boolStyle = { backgroundColor: "#fff0ef", borderColor: "#c24747", color: "#c24747" };
                      }
                    }

                    return (
                      <button
                        key={String(boolVal)}
                        onClick={() => handleAnswerSelect(boolVal)}
                        disabled={selectedAnswer !== null}
                        style={{
                          minHeight: "52px",
                          padding: "0.85rem",
                          borderRadius: "0.75rem",
                          border: "1px solid",
                          fontSize: "1.1rem",
                          fontWeight: "800",
                          cursor: selectedAnswer === null ? "pointer" : "default",
                          ...boolStyle,
                          transition: "all 0.15s ease"
                        }}
                      >
                        {boolVal ? "✅ صحيح" : "❌ خطأ"}
                      </button>
                    );
                  })}
                </div>
              )}

              {/* Explanation & Next Step */}
              {showExplanation && (
                <div style={{ padding: "1rem 1.25rem", backgroundColor: "var(--blue-100)", border: "1px solid #a8d2e2", borderRadius: "0.75rem", marginBottom: "1.5rem" }}>
                  <strong style={{ color: "var(--blue-900)", display: "block", marginBottom: "0.25rem" }}>💡 التوضيح العلمي:</strong>
                  <p style={{ margin: 0, fontSize: "0.9rem", color: "var(--ink-700)", lineHeight: "1.6" }}>
                    {currentQuestion.explanation}
                  </p>
                </div>
              )}

              {selectedAnswer !== null && (
                <div style={{ display: "flex", justifyContent: "flex-end" }}>
                  <button
                    onClick={handleNextQuestion}
                    className="button button-primary"
                    style={{ minHeight: "48px", padding: "0.65rem 1.5rem", fontSize: "0.95rem" }}
                  >
                    {currentIndex + 1 === activeDiagnostic.questions.length ? "عرض النتيجة النهائية 📊" : "السؤال التالي ←"}
                  </button>
                </div>
              )}
            </div>
          ) : (
            /* Diagnostic Report View */
            <div className="calculator-card" style={{ padding: "2rem", textAlign: "center" }}>
              <span style={{ fontSize: "3.5rem" }}>
                {score >= 8 ? "🏆" : score >= 5 ? "📈" : "💪"}
              </span>
              <h2 style={{ color: "var(--blue-900)", margin: "0.5rem 0" }}>تقرير التشخيص لـ {activeDiagnostic.subjectName}</h2>
              <p style={{ fontSize: "1.25rem", fontWeight: "bold", color: "var(--green-600)" }}>
                تحصلت على {score} من {activeDiagnostic.questions.length} أسئلة صحيحة!
              </p>
              <p style={{ color: "var(--ink-700)", maxWidth: "35rem", margin: "0.75rem auto 1.5rem", lineHeight: "1.7" }}>
                {score >= 8
                  ? "ممتاز جداً! لديك قاعدة صلبة ومتقنة للمكتسبات القبلية. أنت جاهز للانطلاق بثقة في البكالوريا."
                  : score >= 5
                  ? "نتيجة طيبة! تمتلك المفاهيم الأساسية ولكن تحتاج لإعادة تنشيط بعض الروابط والقوانين."
                  : "فرصة ممتازة للتدارك! يُنصح بمشاهدة مقاطع المكتسبات القبلية الموصى بها قبل بداية الدروس الرسمية."}
              </p>
              <div style={{ display: "flex", gap: "1rem", justifyContent: "center", flexWrap: "wrap" }}>
                <button onClick={resetQuiz} className="button button-secondary" style={{ minHeight: "48px" }}>
                  إعادة الاختبار 🔄
                </button>
                <button onClick={() => setMode("exercise")} className="button button-primary" style={{ minHeight: "48px" }}>
                  الانتقال للتمرين الشامل 📝
                </button>
              </div>
            </div>
          )}
        </section>
      )}

      {/* MODE 2: COMPREHENSIVE EXERCISE */}
      {mode === "exercise" && (
        <section className="exercise-container" style={{ maxWidth: "62rem" }}>
          <article className="calculator-card" style={{ padding: "1.5rem", marginBottom: "1.5rem" }}>
            <h2 style={{ color: "var(--blue-900)", marginBottom: "1rem", fontSize: "1.25rem" }}>
              {activeDiagnostic.exercise.title}
            </h2>
            <div style={{ whiteSpace: "pre-line", color: "var(--ink-900)", lineHeight: "1.8", fontSize: "0.98rem", marginBottom: "1.5rem" }}>
              {activeDiagnostic.exercise.statement}
            </div>

            <button
              onClick={() => setShowSolution((prev) => !prev)}
              className="button button-secondary"
              style={{ minHeight: "48px", width: "100%", justifyContent: "center", fontSize: "0.95rem" }}
            >
              {showSolution ? "إخفاء الحل النموذجي 👁️" : "عرض الحل النموذجي المفصل 🔑"}
            </button>

            {showSolution && (
              <div style={{ marginTop: "1.5rem", padding: "1.25rem", backgroundColor: "var(--green-100)", border: "1px solid var(--green-600)", borderRadius: "0.85rem" }}>
                <h3 style={{ color: "var(--green-800)", marginTop: 0, marginBottom: "0.75rem", fontSize: "1.1rem" }}>
                  🔑 الحل النموذجي والتنقيط البيداغوجي:
                </h3>
                <div style={{ whiteSpace: "pre-line", color: "var(--green-800)", lineHeight: "1.8", fontSize: "0.95rem" }}>
                  {activeDiagnostic.exercise.solution}
                </div>
              </div>
            )}
          </article>
        </section>
      )}
    </AppShell>
  );
}
```
